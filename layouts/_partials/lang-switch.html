{{- /* Language Switcher — globe icon button with pure CSS dropdown */ -}} {{- /* Only renders when site has 2+ languages configured */ -}} {{- if hugo.IsMultilingual -}} {{- /* Resolve alternate-language link(s). Normal pages use .Translations; series term pages have different slugs per language — Hugo can't auto-link them, so resolve via the first post's translation: its series front matter names the equivalent series in the other language. */ -}} {{- $alts := slice -}} {{- if .Page.IsTranslated }} {{- range .Page.Translations }} {{- $alts = $alts | append (dict "url" .RelPermalink "lang" .Lang "name" .Language.Label) }} {{- end }} {{- else if eq .Type "series" }} {{- $currentLang := .Page.Lang -}} {{- range first 1 .Pages }} {{- range where .Translations "Lang" "ne" $currentLang }} {{- $otherLang := .Lang -}} {{- $otherLangName := .Language.Label -}} {{- $otherRelPermalink := .RelPermalink -}} {{- range first 1 (.Params.series | default slice) }} {{- $slug := . | urlize -}} {{- /* Detect language prefix from translated post URL: //... means non-default */ -}} {{- $hasPrefix := hasPrefix $otherRelPermalink (printf "/%s/" $otherLang) -}} {{- $url := cond $hasPrefix (printf "/%s/series/%s/" $otherLang $slug) (printf "/series/%s/" $slug) -}} {{- $alts = $alts | append (dict "url" $url "lang" $otherLang "name" $otherLangName) -}} {{- end -}} {{- end -}} {{- end -}} {{- end }} {{- /* Button label names the *target* language in its own script, so a reader who can't read the current language still recognizes the switch option (an English speaker on a Chinese page sees "English", not "中文"). */ -}} {{- $buttonLabel := .Page.Language.Label -}} {{- if $alts }} {{- $buttonLabel = (index $alts 0).name -}} {{- end }}
{{- with $alts }}
{{ $.Page.Language.Label }} {{- range . }} {{ .name }} {{- end }}
{{- end }}
{{- end -}}
